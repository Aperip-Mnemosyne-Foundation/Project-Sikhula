psEcon_scanMap = {
	# Owned land describes society, not the unfinished building layer.
	# Rural share and wasteland feed food and unemployment only.
	set_variable = { psEcon_mapPopulationK = 0 }
	set_variable = { psEcon_mapRuralK = 0 }
	set_variable = { psEcon_mapWasteK = 0 }
	set_variable = { psEcon_controlledCount = 0 }
	set_variable = { psEcon_ownedSeen = num_owned_states }

	every_owned_state = {
		set_temp_variable = { popK = state_population_k }
		add_to_variable = { PREV.psEcon_mapPopulationK = popK }
		if = {
			limit = { is_controlled_by = PREV }
			add_to_variable = { PREV.psEcon_controlledCount = 1 }
		}

		set_temp_variable = { ruralWeight = 0 }
		set_temp_variable = { wasteWeight = 0 }
		if = {
			limit = { has_state_category = pastoral }
			set_temp_variable = { ruralWeight = 0.75 }
		}
		else_if = {
			limit = { has_state_category = rural }
			set_temp_variable = { ruralWeight = 0.85 }
		}
		else_if = {
			limit = { has_state_category = village }
			set_temp_variable = { ruralWeight = 0.70 }
		}
		else_if = {
			limit = { has_state_category = town }
			set_temp_variable = { ruralWeight = 0.35 }
		}
		else_if = {
			limit = { has_state_category = large_town }
			set_temp_variable = { ruralWeight = 0.22 }
		}
		else_if = {
			limit = { has_state_category = wasteland }
			set_temp_variable = { wasteWeight = 1 }
		}
		else_if = {
			limit = { has_state_category = enclave }
			set_temp_variable = { wasteWeight = 0.35 }
		}

		set_temp_variable = { weighted = popK }
		multiply_temp_variable = { weighted = ruralWeight }
		add_to_variable = { PREV.psEcon_mapRuralK = weighted }
		set_temp_variable = { weighted = popK }
		multiply_temp_variable = { weighted = wasteWeight }
		add_to_variable = { PREV.psEcon_mapWasteK = weighted }
	}

	set_temp_variable = { popK = psEcon_mapPopulationK }
	if = {
		limit = { check_variable = { popK < 1 } }
		set_temp_variable = { popK = 1 }
	}
	set_variable = { psEcon_mapPopulation = psEcon_mapPopulationK }
	divide_variable = { psEcon_mapPopulation = 1000 }
	set_variable = { psEcon_ruralShare = psEcon_mapRuralK }
	divide_variable = { psEcon_ruralShare = popK }
	set_variable = { psEcon_wasteShare = psEcon_mapWasteK }
	divide_variable = { psEcon_wasteShare = popK }
	set_variable = { psEcon_control = psEcon_controlledCount }
	if = {
		limit = { check_variable = { psEcon_ownedSeen > 0 } }
		divide_variable = { psEcon_control = psEcon_ownedSeen }
	}
	else = {
		set_variable = { psEcon_control = 1 }
	}
	clamp_variable = { var = psEcon_control min = 0.20 max = 1 }
	set_variable = { psEcon_scanWait = 0 }
}
