EC_cre = {
	#计算实体经济
	set_variable = {
		ROOT.ECSB = ROOT.num_of_civilian_factories
	}
	set_variable = {
		ROOT.ECRE = ROOT.ECSB
	}
	if = {
		limit = { 
			NOT = { has_variable = ROOT.ECPB }
		}
		set_variable = {
			ROOT.ECPB = 0
		}
	}
	if = {
		limit = { 
			has_variable = ROOT.ECPB_mem
		}
		set_variable = {
			ROOT.ECPB_cbef = ROOT.ECPB_mem
		}
	}
	else = {
		set_variable = {
			ROOT.ECPB_cbef = 0
		}
	}
	subtract_from_variable = {
		ROOT.ECPB_cbef = ROOT.ECPB
	}
	multiply_variable = {
		ROOT.ECPB_cbef = -1
	}
	add_offsite_building = {
		type = industrial_complex
		level = ROOT.ECPB_cbef
	}
	subtract_from_variable = {
		ROOT.ECSB = ROOT.ECPB
	}
	set_variable = {
		ROOT.ECPB_mem = ROOT.ECPB
	}
	set_variable = {
		ROOT.ECRE_SHOW = ROOT.ECRE
	}
}

EC_ceg = {
	#计算经济增长速度（人均）（每月计算)
	if = {
		limit = { 
			check_variable = {
				ROOT.EC_PE_A^num = 500
			}
		}
        remove_from_array = {
            array = ROOT.EC_PE_A
            index = 0
        }
	}
	add_to_array = {
		array = ROOT.EC_PE_A
		value = ROOT.EC_PE
	}
	set_variable = {
		ROOT.EC_PE_num = ROOT.EC_PE_A^num
	}
	set_variable = {
		ROOT.EC_PE_A_ibef = ROOT.EC_PE_num
	}
	add_to_variable = {
		ROOT.EC_PE_A_ibef = -1
	}
	set_variable = {
		ROOT.EC_PE_cnow = ROOT.EC_PE_A^ROOT.EC_PE_A_ibef
	}
	if = {
		limit = {
            check_variable = {
				ROOT.EC_PE_A_ibef > 0
			}
		}
        add_to_variable = {
		    ROOT.EC_PE_A_ibef = -1
	    }
	}
	set_variable = {
		ROOT.EC_PE_cbef = ROOT.EC_PE_A^ROOT.EC_PE_A_ibef
	}
	set_variable = {
		ROOT.EC_PE_c1 = ROOT.EC_PE_cnow
	}
	subtract_from_variable = {
		ROOT.EC_PE_c1 = ROOT.EC_PE_cbef
	}
	multiply_variable = {
		ROOT.EC_PE_c1 = 10000
	}
	if = {
		limit = { 
			check_variable = {
				ROOT.EC_PE_cbef = 0
			}
		}
		divide_variable = {
		    ROOT.EC_PE_c1 = ROOT.EC_PE_cnow
	    }
	}
	else = {
		divide_variable = {
		    ROOT.EC_PE_c1 = ROOT.EC_PE_cbef
	    }
	}
	set_variable = {
		ROOT.ECEG = ROOT.EC_PE_c1
	}
	set_variable = {
		ROOT.ECEG_A_imax = ROOT.ECEG_A^num
	}
	if = {
		limit = { 
			check_variable = {
				ROOT.ECEG_A_imax = 500
			}
		}
        remove_from_array = {
            array = ROOT.ECEG_A
            index = 0
        }
	}
	add_to_array = {
		array = ROOT.ECEG_A
		value = ROOT.ECEG
	}
}
