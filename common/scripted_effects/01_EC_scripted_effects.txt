# Build the frame list for the old total-output sparkline.
ECOA_graph = {
	clear_array = ROOT.ECOA_y_label
	set_variable = { ROOT.ECOA_min = ROOT.ECOA_numbers^0 }
	set_variable = { ROOT.ECOA_max = ROOT.ECOA_numbers^0 }
	for_each_loop = {
		array = ROOT.ECOA_numbers
		value = v
		if = {
			limit = {
                check_variable = {
					v > ROOT.ECOA_max
				}
			}
			set_variable = {
				ROOT.ECOA_max = v
			}
		}
		if = {
			limit = { 
				check_variable = {
					v < ROOT.ECOA_min
				}
			}
			set_variable = {
				ROOT.ECOA_min = v
			}
		}
	}
	set_variable = {
		ROOT.ECOA_min_show = ROOT.ECOA_min
	}
	set_variable = {
		ROOT.ECOA_max_show = ROOT.ECOA_max
	}
	multiply_variable = { ROOT.ECOA_max = 1.1 }
	divide_variable = { ROOT.ECOA_min = 1.1 }
	set_variable = { ROOT.ECOA_range = ROOT.ECOA_max }
	subtract_from_variable = { ROOT.ECOA_range = ROOT.ECOA_min }
	set_variable = { ROOT.ECOA_display_increment = ROOT.ECOA_range }
	divide_variable = { ROOT.ECOA_display_increment = 4 }
	set_variable = { ROOT.ECOA_band_display = ROOT.ECOA_min }
	for_loop_effect = {
		end = 5
		add_to_array = { array = ROOT.ECOA_y_label value = ROOT.ECOA_band_display index = 0 }
		add_to_variable = { ROOT.ECOA_band_display = ROOT.ECOA_display_increment }
	}
	divide_variable = { ROOT.ECOA_range = 100 }
	clear_array = ROOT.ECOA_graph_frames
	for_each_loop = {
		array = ROOT.ECOA_numbers
		value = v
		index = i
		set_variable = {
			ROOT.ECOA_size = ROOT.ECOA_numbers^num
		}
		add_to_variable = {
			ROOT.ECOA_size = -1
		}
		if = {
			limit = { 
				NOT = {
					check_variable = {
						i = ROOT.ECOA_size
					}
				}
			}
			set_variable = {
			    ROOT.ECOA_n = v
		    }
		    set_variable = {
			    ROOT.ECOA_lidx = i
		    }
		    add_to_variable = {
		    	ROOT.ECOA_lidx = 1
		    }
		    set_variable = {
		    	ROOT.ECOA_l = ROOT.ECOA_numbers^ECOA_lidx
		    }
			subtract_from_variable = { ROOT.ECOA_n = ROOT.ECOA_min }
			subtract_from_variable = { ROOT.ECOA_l = ROOT.ECOA_min }
			divide_variable = { ROOT.ECOA_n = ROOT.ECOA_range }
			divide_variable = { ROOT.ECOA_l = ROOT.ECOA_range }
			round_variable = ROOT.ECOA_n 
			round_variable = ROOT.ECOA_l
			set_variable = { ROOT.ECOA_endpoints = ROOT.ECOA_n }
			multiply_variable = { ROOT.ECOA_endpoints = 1000 }
			add_to_variable = { ROOT.ECOA_endpoints = ROOT.ECOA_l }
			add_to_array = { ROOT.ECOA_graph_frames = ROOT.ECOA_endpoints }
		}
	}
	
}
