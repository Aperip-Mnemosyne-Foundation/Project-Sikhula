# World prices and standing laws, copied into plain variables for the rest of the month.

# Standing laws are evaluated in mtth:. This block only copies the bite.
psEcon_readWorld = {
	set_variable = { psEcon_atWar = 0 }
	if = {
		limit = { has_war = yes }
		set_variable = { psEcon_atWar = 1 }
	}

	set_variable = { psEcon_taxEnforce = mtth:psLawMtth_taxEnforce }
	set_variable = { psLaw_showTax = mtth:psLawMtth_taxTake }
	set_variable = { psEcon_laborBite = mtth:psLawMtth_laborBite }
	set_variable = { psLaw_showLabor = psEcon_laborBite }
	set_variable = { psEcon_tradeLevy = mtth:psLawMtth_tradeLevy }
	set_variable = { psLaw_showLevy = psEcon_tradeLevy }
	set_variable = { psEcon_capitalSeal = mtth:psLawMtth_capitalSeal }
	set_variable = { psLaw_showSeal = psEcon_capitalSeal }
	set_variable = { psEcon_contractTrust = mtth:psLawMtth_contractTrust }
	set_variable = { psLaw_showTrust = psEcon_contractTrust }
	set_variable = { psEcon_soePrivilege = mtth:psLawMtth_soePrivilege }
	set_variable = { psLaw_showSoe = psEcon_soePrivilege }
	set_variable = { psLaw_showFactory = mtth:psLawMtth_laborFactory }
	set_variable = { psLaw_showResearch = mtth:psLawMtth_researchFactor }
	set_variable = { psLaw_showBuild = mtth:psLawMtth_soeBuild }
	set_variable = { psLaw_showWeekly = mtth:psLawMtth_stabilityWeekly }
	clamp_variable = { var = psEcon_tradeOpen min = 0.05 max = 0.95 }
	set_temp_variable = { liquid = psEcon_forex }
	divide_temp_variable = { liquid = 3 }
	clamp_temp_variable = { var = liquid min = 0 max = 1 }
	set_temp_variable = { locked = 1 }
	subtract_from_temp_variable = { locked = liquid }
	multiply_temp_variable = { locked = psEcon_fxPrivilege }
	set_temp_variable = { access = liquid }
	add_to_temp_variable = { access = locked }
	set_variable = { psEcon_fxAccess = access }
	set_temp_variable = { tradeUsed = psEcon_tradeOpen }
	multiply_temp_variable = { tradeUsed = access }
	set_temp_variable = { levyKeep = psEcon_tradeLevy }
	multiply_temp_variable = { levyKeep = 0.50 }
	set_temp_variable = { levyScale = 1 }
	subtract_from_temp_variable = { levyScale = levyKeep }
	clamp_temp_variable = { var = levyScale min = 0.55 max = 1 }
	multiply_temp_variable = { tradeUsed = levyScale }
	set_variable = { psEcon_tradeUsed = tradeUsed }

	set_temp_variable = { raw = modifier@fiscalRevenue_factor }
	divide_temp_variable = { raw = 100 }
	clamp_temp_variable = { var = raw min = -0.30 max = 0.30 }
	set_variable = { psEcon_fiscRev = raw }
	set_temp_variable = { raw = modifier@fiscalExpenditure_factor }
	divide_temp_variable = { raw = 100 }
	clamp_temp_variable = { var = raw min = -0.30 max = 0.30 }
	set_variable = { psEcon_fiscExp = raw }
	set_temp_variable = { raw = modifier@governmentCredibility }
	clamp_temp_variable = { var = raw min = -0.40 max = 0.40 }
	set_variable = { psEcon_ideaGov = raw }
	set_temp_variable = { raw = modifier@economicCredibility }
	clamp_temp_variable = { var = raw min = -0.40 max = 0.40 }
	set_variable = { psEcon_ideaMarket = raw }
	set_temp_variable = { raw = modifier@monetaryValue }
	clamp_temp_variable = { var = raw min = -2 max = 2 }
	set_variable = { psEcon_monetaryAdd = raw }

	set_temp_variable = { raw = modifier@militaryExpenditure_baseDaily }
	clamp_temp_variable = { var = raw min = -0.05 max = 0.05 }
	set_variable = { psEcon_milBase = raw }
	set_temp_variable = { raw = modifier@militaryExpenditure_factor }
	divide_temp_variable = { raw = 100 }
	clamp_temp_variable = { var = raw min = -0.30 max = 0.30 }
	set_variable = { psEcon_milFac = raw }
	set_temp_variable = { raw = modifier@educationExpenditure_baseDaily }
	clamp_temp_variable = { var = raw min = -0.05 max = 0.05 }
	set_variable = { psEcon_eduBase = raw }
	set_temp_variable = { raw = modifier@educationExpenditure_factor }
	divide_temp_variable = { raw = 100 }
	clamp_temp_variable = { var = raw min = -0.30 max = 0.30 }
	set_variable = { psEcon_eduFac = raw }
	set_temp_variable = { raw = modifier@socialWelfareExpenditure_baseDaily }
	clamp_temp_variable = { var = raw min = -0.05 max = 0.05 }
	set_variable = { psEcon_welBase = raw }
	set_temp_variable = { raw = modifier@socialWelfareExpenditure_factor }
	divide_temp_variable = { raw = 100 }
	clamp_temp_variable = { var = raw min = -0.30 max = 0.30 }
	set_variable = { psEcon_welFac = raw }

	# Investment and environment expenditure use the same contract, even without a separate token yet.
	set_variable = { psEcon_invBase = 0 }
	set_variable = { psEcon_invFac = 0 }
	set_temp_variable = { raw = modifier@environmentalDevelopment_baseDaily }
	clamp_temp_variable = { var = raw min = -0.02 max = 0.02 }
	set_variable = { psEcon_envDrift = raw }
	set_variable = { psEcon_envBase = 0 }
	set_variable = { psEcon_envFac = 0 }

	# baseDaily 0.01 is one point of the related annual flow. Factors are percentage points.
	set_temp_variable = { foodBase = modifier@agriculturalDevelopment_baseDaily }
	set_temp_variable = { extra = modifier@forestryDevelopment_baseDaily }
	add_to_temp_variable = { foodBase = extra }
	set_temp_variable = { extra = modifier@fisheriesDevelopment_baseDaily }
	add_to_temp_variable = { foodBase = extra }
	set_temp_variable = { extra = modifier@livestockIndustrialDevelopment_baseDaily }
	add_to_temp_variable = { foodBase = extra }
	clamp_temp_variable = { var = foodBase min = -0.04 max = 0.04 }
	set_temp_variable = { foodFac = modifier@agriculturalDevelopment_factor }
	set_temp_variable = { extra = modifier@forestryDevelopment_factor }
	add_to_temp_variable = { foodFac = extra }
	set_temp_variable = { extra = modifier@fisheriesDevelopment_factor }
	add_to_temp_variable = { foodFac = extra }
	set_temp_variable = { extra = modifier@livestockIndustrialDevelopment_factor }
	add_to_temp_variable = { foodFac = extra }
	divide_temp_variable = { foodFac = 400 }
	clamp_temp_variable = { var = foodFac min = -0.30 max = 0.30 }
	add_to_temp_variable = { foodFac = 1 }
	multiply_temp_variable = { foodBase = foodFac }
	clamp_temp_variable = { var = foodBase min = -0.06 max = 0.06 }
	set_variable = { psEcon_chFood = foodBase }

	set_temp_variable = { indBase = modifier@lightIndustrialDevelopment_baseDaily }
	set_temp_variable = { extra = modifier@heavyIndustrialDevelopment_baseDaily }
	add_to_temp_variable = { indBase = extra }
	clamp_temp_variable = { var = indBase min = -0.02 max = 0.02 }
	set_variable = { psEcon_chIndustry = indBase }

	set_temp_variable = { techBase = modifier@highTechnologyIndustrialDevelopment_baseDaily }
	set_temp_variable = { techFac = modifier@highTechnologyIndustrialDevelopment_factor }
	divide_temp_variable = { techFac = 100 }
	clamp_temp_variable = { var = techFac min = -0.30 max = 0.30 }
	add_to_temp_variable = { techFac = 1 }
	multiply_temp_variable = { techBase = techFac }
	set_temp_variable = { extra = modifier@highTechnologyToints_baseDaily }
	add_to_temp_variable = { techBase = extra }
	set_temp_variable = { info = modifier@informatization_level }
	divide_temp_variable = { info = 5000 }
	add_to_temp_variable = { techBase = info }
	clamp_temp_variable = { var = techBase min = -0.02 max = 0.02 }
	set_variable = { psEcon_chTech = techBase }
}
