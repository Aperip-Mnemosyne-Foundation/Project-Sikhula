# Monthly score. These effects only write variables. psEcon_publish is what reaches the game.

psEcon_calcBalances = {

	set_temp_variable = { gov = psEcon_govStock }
	add_to_temp_variable = { gov = psEcon_ideaGov }
	clamp_temp_variable = { var = gov min = 0.05 max = 0.95 }
	set_variable = { psEcon_govNow = gov }

	add_to_temp_variable = { market = psEcon_ideaMarket }

	add_to_temp_variable = { stance = psEcon_monetaryAdd }
	clamp_temp_variable = { var = stance min = -2 max = 2 }
	set_variable = { psEcon_stanceUsed = stance }

	add_to_temp_variable = { shareSum = psEcon_eduPaid }
	add_to_temp_variable = { shareSum = psEcon_welPaid }
	add_to_temp_variable = { shareSum = psEcon_invPaid }
	add_to_temp_variable = { shareSum = psEcon_envPaid }
	if = {
		limit = { check_variable = { shareSum < 0.01 } }
		set_temp_variable = { shareSum = 1 }
	}
	set_variable = { psEcon_milS = psEcon_milPaid }
	divide_variable = { psEcon_milS = shareSum }
	set_variable = { psEcon_eduS = psEcon_eduPaid }
	divide_variable = { psEcon_eduS = shareSum }
	set_variable = { psEcon_welS = psEcon_welPaid }
	divide_variable = { psEcon_welS = shareSum }
	set_variable = { psEcon_invShare = psEcon_invPaid }
	divide_variable = { psEcon_invShare = shareSum }
	set_variable = { psEcon_envS = psEcon_envPaid }
	divide_variable = { psEcon_envS = shareSum }

	set_temp_variable = { milF = psEcon_spendPaid }
	multiply_temp_variable = { milF = psEcon_milS }
	add_to_temp_variable = { milF = psEcon_milBase }
	set_temp_variable = { scale = psEcon_milFac }
	add_to_temp_variable = { scale = 1 }
	multiply_temp_variable = { milF = scale }
	clamp_temp_variable = { var = milF min = 0 max = 0.28 }
	set_temp_variable = { eduF = psEcon_spendPaid }
	multiply_temp_variable = { eduF = psEcon_eduS }
	add_to_temp_variable = { eduF = psEcon_eduBase }
	set_temp_variable = { scale = psEcon_eduFac }
	add_to_temp_variable = { scale = 1 }
	multiply_temp_variable = { eduF = scale }
	clamp_temp_variable = { var = eduF min = 0 max = 0.28 }
	set_temp_variable = { welF = psEcon_spendPaid }
	multiply_temp_variable = { welF = psEcon_welS }
	add_to_temp_variable = { welF = psEcon_welBase }
	set_temp_variable = { scale = psEcon_welFac }
	add_to_temp_variable = { scale = 1 }
	multiply_temp_variable = { welF = scale }
	clamp_temp_variable = { var = welF min = 0 max = 0.28 }
	set_temp_variable = { invF = psEcon_spendPaid }
	multiply_temp_variable = { invF = psEcon_invShare }
	add_to_temp_variable = { invF = psEcon_invBase }
	set_temp_variable = { scale = psEcon_invFac }
	add_to_temp_variable = { scale = 1 }
	multiply_temp_variable = { invF = scale }
	clamp_temp_variable = { var = invF min = 0 max = 0.28 }
	set_temp_variable = { envF = psEcon_spendPaid }
	multiply_temp_variable = { envF = psEcon_envS }
	add_to_temp_variable = { envF = psEcon_envBase }
	set_temp_variable = { scale = psEcon_envFac }
	add_to_temp_variable = { scale = 1 }
	multiply_temp_variable = { envF = scale }
	clamp_temp_variable = { var = envF min = 0 max = 0.28 }
	set_temp_variable = { useful = milF }
	add_to_temp_variable = { useful = eduF }
	add_to_temp_variable = { useful = welF }
	add_to_temp_variable = { useful = invF }
	add_to_temp_variable = { useful = envF }
	set_temp_variable = { warLoad = psEcon_atWar }
	multiply_temp_variable = { warLoad = 0.12 }
	add_to_temp_variable = { warLoad = 1 }
	set_temp_variable = { spendActual = useful }
	set_temp_variable = { govInv = invF }
	set_temp_variable = { leak = psEcon_fiscExp }
	add_to_temp_variable = { leak = 1 }
	multiply_temp_variable = { spendActual = leak }
	if = {
		limit = { check_variable = { psEcon_fiscExp < 0 } }
		multiply_temp_variable = { govInv = leak }
	}
	multiply_temp_variable = { spendActual = warLoad }
	multiply_temp_variable = { govInv = warLoad }
	if = {
		limit = {
			check_variable = { psEcon_therapyLeft > 0 }
			check_variable = { psEcon_spendPaid < psEcon_spendNeutral }
		}
		set_temp_variable = { leaked = psEcon_spendNeutral }
		subtract_from_temp_variable = { leaked = psEcon_spendPaid }
		set_temp_variable = { weak = 1 }
		subtract_from_temp_variable = { weak = gov }
		multiply_temp_variable = { leaked = weak }
		multiply_temp_variable = { leaked = 0.55 }
		add_to_temp_variable = { spendActual = leaked }
	}
	# Save the quote before cover. Unpaid budget does not deliver the construction it promised.
	set_variable = { psEcon_spendQuote = spendActual }
	set_temp_variable = { cover = psEcon_cover }
	clamp_temp_variable = { var = cover min = 0.40 max = 1 }
	multiply_temp_variable = { spendActual = cover }
	multiply_temp_variable = { govInv = cover }
	multiply_variable = { psEcon_milS = cover }
	multiply_variable = { psEcon_eduS = cover }
	multiply_variable = { psEcon_welS = cover }
	multiply_variable = { psEcon_invShare = cover }
	multiply_variable = { psEcon_envS = cover }
	set_variable = { psEcon_spendActual = spendActual }
	set_variable = { psEcon_govInv = govInv }

	set_temp_variable = { foodEnd = psEcon_ruralShare }
	multiply_temp_variable = { foodEnd = 0.55 }
	add_to_temp_variable = { foodEnd = 0.78 }
	set_temp_variable = { wasteHit = psEcon_wasteShare }
	multiply_temp_variable = { wasteHit = 0.50 }
	subtract_from_temp_variable = { foodEnd = wasteHit }
	clamp_temp_variable = { var = foodEnd min = 0.50 max = 1.45 }
	set_temp_variable = { foodSupply = psEcon_environment }
	multiply_temp_variable = { foodSupply = 0.30 }
	add_to_temp_variable = { foodSupply = 0.85 }
	set_temp_variable = { controlBit = psEcon_control }
	multiply_temp_variable = { controlBit = 0.30 }
	add_to_temp_variable = { controlBit = 0.70 }
	multiply_temp_variable = { foodSupply = controlBit }
	multiply_temp_variable = { foodSupply = foodEnd }
	set_temp_variable = { warKeep = 1 }
	set_temp_variable = { warFood = psEcon_atWar }
	multiply_temp_variable = { warFood = 0.06 }
	subtract_from_temp_variable = { warKeep = warFood }
	multiply_temp_variable = { foodSupply = warKeep }
	set_temp_variable = { foodIdea = psEcon_chFood }
	add_to_temp_variable = { foodIdea = 1 }
	multiply_temp_variable = { foodSupply = foodIdea }
	set_temp_variable = { grainGap = global.psEcon_grain }
	subtract_from_temp_variable = { grainGap = 1 }
	clamp_temp_variable = { var = grainGap min = 0 max = 2 }
	multiply_temp_variable = { grainGap = 0.04 }
	set_temp_variable = { grainKeep = 1 }
	subtract_from_temp_variable = { grainKeep = grainGap }
	multiply_temp_variable = { foodSupply = grainKeep }
	set_temp_variable = { shockKeep = 1 }
	subtract_from_temp_variable = { shockKeep = psEcon_shockFood }
	multiply_temp_variable = { foodSupply = shockKeep }
	# Emergency grain arrives after the harvest shock. It is a stock, not a free harvest.
	add_to_temp_variable = { foodSupply = psEcon_foodRelief }
	add_to_temp_variable = { foodSupply = psInd_foodAdd }
	set_temp_variable = { diet = psEcon_income }
	divide_temp_variable = { diet = 30 }
	clamp_temp_variable = { var = diet min = 0 max = 1.2 }
	multiply_temp_variable = { diet = 0.08 }
	add_to_temp_variable = { diet = 0.92 }
	set_temp_variable = { foodBalance = foodSupply }
	divide_temp_variable = { foodBalance = diet }
	set_variable = { psEcon_foodBalance = foodBalance }
	set_temp_variable = { foodEffective = foodBalance }
	if = {
		limit = { check_variable = { foodBalance < 1 } }
		set_temp_variable = { room = global.psEcon_grain }
		subtract_from_temp_variable = { room = 1 }
		multiply_temp_variable = { room = 0.25 }
		set_temp_variable = { roomKeep = 1.15 }
		subtract_from_temp_variable = { roomKeep = room }
		clamp_temp_variable = { var = roomKeep min = 0.40 max = 1.15 }
		set_temp_variable = { gap = 1 }
		subtract_from_temp_variable = { gap = foodBalance }
		multiply_temp_variable = { gap = psEcon_tradeUsed }
		multiply_temp_variable = { gap = roomKeep }
		add_to_temp_variable = { foodEffective = gap }
	}
	set_temp_variable = { foodStress = 0.98 }
	subtract_from_temp_variable = { foodStress = foodEffective }
	divide_temp_variable = { foodStress = 0.20 }
	clamp_temp_variable = { var = foodStress min = 0 max = 1 }
	set_variable = { psEcon_foodStress = foodStress }
	set_variable = { psEcon_foodEffective = foodEffective }

	set_temp_variable = { oilFace = global.psEcon_oil }
	add_to_temp_variable = { oilFace = psEcon_shockOil }
	set_variable = { psEcon_oilFace = oilFace }
	set_temp_variable = { domestic = psEcon_control }
	multiply_temp_variable = { domestic = 0.25 }
	add_to_temp_variable = { domestic = 0.75 }
	multiply_temp_variable = { domestic = psEcon_energy }
	add_to_temp_variable = { domestic = psInd_energyAdd }
	set_temp_variable = { importFactor = oilFace }
	subtract_from_temp_variable = { importFactor = 1 }
	multiply_temp_variable = { importFactor = 0.35 }
	set_temp_variable = { importKeep = 1.20 }
	subtract_from_temp_variable = { importKeep = importFactor }
	clamp_temp_variable = { var = importKeep min = 0.25 max = 1.20 }
	set_temp_variable = { importNeed = 1.05 }
	subtract_from_temp_variable = { importNeed = domestic }
	clamp_temp_variable = { var = importNeed min = 0 max = 2 }
	multiply_temp_variable = { importNeed = psEcon_tradeUsed }
	multiply_temp_variable = { importNeed = importKeep }
	set_temp_variable = { energyCover = domestic }
	add_to_temp_variable = { energyCover = importNeed }
	clamp_temp_variable = { var = energyCover min = 0.05 max = 2.2 }
	set_temp_variable = { energyStress = 0.95 }
	subtract_from_temp_variable = { energyStress = energyCover }
	divide_temp_variable = { energyStress = 0.35 }
	clamp_temp_variable = { var = energyStress min = 0 max = 1 }
	set_variable = { psEcon_energyCover = energyCover }
	set_variable = { psEcon_energyStress = energyStress }
}
# Shock therapy is a real package: it hurts immediately, and only a capable state gets the recovery.
psEcon_calcDemand = {
	set_temp_variable = { gov = psEcon_govNow }
	set_temp_variable = { stance = psEcon_stanceUsed }
	set_variable = { psEcon_therapyPain = 0 }
	set_variable = { psEcon_therapyChaos = 0 }
	set_variable = { psEcon_therapyDisinflation = 0 }
	set_variable = { psEcon_therapyRecovery = 0 }
	set_variable = { psEcon_therapyDiscipline = 0 }
	set_variable = { psEcon_therapyJobs = 0 }
	if = {
		limit = { check_variable = { psEcon_therapyLeft > 0 } }
		set_temp_variable = { spendCut = psEcon_spendNeutral }
		subtract_from_temp_variable = { spendCut = psEcon_spendPaid }
		divide_temp_variable = { spendCut = 0.08 }
		clamp_temp_variable = { var = spendCut min = 0 max = 1 }
		set_temp_variable = { moneyTight = stance }
		multiply_temp_variable = { moneyTight = -1 }
		divide_temp_variable = { moneyTight = 2 }
		clamp_temp_variable = { var = moneyTight min = 0 max = 1 }
		set_temp_variable = { discipline = spendCut }
		add_to_temp_variable = { discipline = moneyTight }
		multiply_temp_variable = { discipline = 0.5 }
		set_variable = { psEcon_therapyDiscipline = discipline }
		set_temp_variable = { early = 0.25 }
		if = {
			limit = { check_variable = { psEcon_therapyElapsed < 12 } }
			set_temp_variable = { early = 1 }
		}
		else_if = {
			limit = { check_variable = { psEcon_therapyElapsed < 24 } }
			set_temp_variable = { early = 0.55 }
		}
		set_temp_variable = { pain = discipline }
		multiply_temp_variable = { pain = 0.020 }
		add_to_temp_variable = { pain = 0.030 }
		set_temp_variable = { capacityKeep = gov }
		multiply_temp_variable = { capacityKeep = 0.45 }
		set_temp_variable = { keep = 1.10 }
		subtract_from_temp_variable = { keep = capacityKeep }
		multiply_temp_variable = { pain = keep }
		multiply_temp_variable = { pain = early }
		set_variable = { psEcon_therapyPain = pain }
		set_temp_variable = { weakState = 1 }
		subtract_from_temp_variable = { weakState = gov }
		set_temp_variable = { chaos = discipline }
		add_to_temp_variable = { chaos = 0.35 }
		multiply_temp_variable = { chaos = weakState }
		multiply_temp_variable = { chaos = 0.030 }
		multiply_temp_variable = { chaos = early }
		set_variable = { psEcon_therapyChaos = chaos }
		set_temp_variable = { disinflation = discipline }
		multiply_temp_variable = { disinflation = 0.014 }
		add_to_temp_variable = { disinflation = 0.004 }
		multiply_temp_variable = { disinflation = gov }
		if = {
			limit = { check_variable = { psEcon_therapyElapsed < 6 } }
			multiply_temp_variable = { disinflation = 0.35 }
		}
		set_variable = { psEcon_therapyDisinflation = disinflation }
		if = {
			limit = {
				check_variable = { psEcon_therapyElapsed > 15.9 }
				check_variable = { psEcon_inflation < 0.09 }
				check_variable = { gov > 0.58 }
				check_variable = { discipline > 0.55 }
			}
			set_temp_variable = { recovery = psEcon_therapyElapsed }
			subtract_from_temp_variable = { recovery = 16 }
			divide_temp_variable = { recovery = 10 }
			clamp_temp_variable = { var = recovery min = 0 max = 1 }
			multiply_temp_variable = { recovery = 0.012 }
			set_variable = { psEcon_therapyRecovery = recovery }
		}
		set_temp_variable = { jobs = pain }
		multiply_temp_variable = { jobs = 0.85 }
		set_variable = { psEcon_therapyJobs = jobs }
	}

	set_temp_variable = { demand = 0.985 }
	add_to_temp_variable = { demand = psEcon_demandMood }
	set_temp_variable = { spendGap = psEcon_spendPaid }
	subtract_from_temp_variable = { spendGap = psEcon_spendNeutral }
	multiply_temp_variable = { spendGap = 0.42 }
	add_to_temp_variable = { demand = spendGap }
	set_temp_variable = { stanceBit = psEcon_stanceUsed }
	multiply_temp_variable = { stanceBit = 0.028 }
	add_to_temp_variable = { demand = stanceBit }
	set_temp_variable = { warBit = psEcon_atWar }
	multiply_temp_variable = { warBit = 0.03 }
	add_to_temp_variable = { demand = warBit }
	set_temp_variable = { taxGap = psEcon_taxPaid }
	subtract_from_temp_variable = { taxGap = psEcon_taxNeutral }
	multiply_temp_variable = { taxGap = 0.32 }
	subtract_from_temp_variable = { demand = taxGap }
	add_to_temp_variable = { demand = psEcon_shockDemand }
	set_temp_variable = { laborBit = psEcon_laborBite }
	subtract_from_temp_variable = { laborBit = 0.35 }
	multiply_temp_variable = { laborBit = 0.025 }
	subtract_from_temp_variable = { demand = laborBit }
	if = {
		limit = { has_country_flag = psEcon_bursting }
		set_temp_variable = { burstDemand = psEcon_bubble }
		multiply_temp_variable = { burstDemand = 0.028 }
		add_to_temp_variable = { burstDemand = 0.022 }
		subtract_from_temp_variable = { demand = burstDemand }
	}
	subtract_from_temp_variable = { demand = psEcon_therapyPain }
	clamp_temp_variable = { var = demand min = 0.84 max = 1.05 }
	set_variable = { psEcon_util = demand }

	set_temp_variable = { uInstant = 0.040 }
	add_to_temp_variable = { uInstant = psEcon_friction }
	set_temp_variable = { ruralU = psEcon_ruralShare }
	multiply_temp_variable = { ruralU = 0.028 }
	add_to_temp_variable = { uInstant = ruralU }
	set_temp_variable = { slack = 0.985 }
	subtract_from_temp_variable = { slack = psEcon_util }
	multiply_temp_variable = { slack = 0.42 }
	add_to_temp_variable = { uInstant = slack }
	set_temp_variable = { eduU = psEcon_education }
	subtract_from_temp_variable = { eduU = 0.50 }
	multiply_temp_variable = { eduU = 0.018 }
	subtract_from_temp_variable = { uInstant = eduU }
	set_temp_variable = { laborU = psEcon_laborBite }
	subtract_from_temp_variable = { laborU = 0.35 }
	multiply_temp_variable = { laborU = 0.045 }
	add_to_temp_variable = { uInstant = laborU }
	add_to_temp_variable = { uInstant = psEcon_therapyJobs }
	# State firms hoard labor. The stock lives in the industry file.
	subtract_from_temp_variable = { uInstant = psInd_jobHoard }
	set_temp_variable = { welU = psEcon_welS }
	subtract_from_temp_variable = { welU = 0.22 }
	clamp_temp_variable = { var = welU min = 0 max = 1 }
	multiply_temp_variable = { welU = 0.030 }
	subtract_from_temp_variable = { uInstant = welU }
	if = {
		limit = { has_country_flag = psEcon_bursting }
		set_temp_variable = { burstU = psEcon_bubble }
		multiply_temp_variable = { burstU = 0.025 }
		add_to_temp_variable = { uInstant = burstU }
	}
	clamp_temp_variable = { var = uInstant min = 0.025 max = 0.28 }
	set_variable = { psEcon_uInstant = uInstant }
	set_variable = { psEcon_uUsed = psEcon_unemployment }
	if = {
		limit = { check_variable = { psEcon_uUsed < 0.001 } }
		set_variable = { psEcon_uUsed = psEcon_uInstant }
	}
}

	multiply_temp_variable = { sheetPot = 0.010 }
	subtract_from_temp_variable = { potential = sheetPot }

	subtract_from_temp_variable = { heat = potential }
	subtract_from_temp_variable = { heat = 0.005 }
	clamp_temp_variable = { var = heat min = 0 max = 0.045 }

		limit = {
			check_variable = { psEcon_util > 1.02 }
			check_variable = { psEcon_stanceUsed > 0 }

	clamp_temp_variable = { var = volPen min = 0 max = 1 }
	set_temp_variable = { infPen = psEcon_inflation }
