psLaw_shiftSupport = {
	# psLaw_inputLaw: 1 tax, 2 labor, 3 tariff, 4 capital, 5 contract, 6 state-firm privilege.
	if = {
		limit = { check_variable = { psLaw_inputLaw = 1 } }
		add_to_variable = { psLaw_taxSupport = psLaw_inputStep }
		clamp_variable = { var = psLaw_taxSupport min = 0.05 max = 0.95 }
	}
	else_if = {
		limit = { check_variable = { psLaw_inputLaw = 2 } }
