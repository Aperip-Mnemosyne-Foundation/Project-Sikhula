# Statutory codes and the support that decides whether they bite.
# Codes and support are history data. Events call psLaw_bump* with psLaw_inputStep.

psLaw_fallback = {
	if = {
		limit = { NOT = { has_variable = psLaw_taxCode } }
		set_variable = { psLaw_taxCode = 0.24 }
		set_variable = { psLaw_taxSupport = 0.50 }
		set_variable = { psLaw_laborCode = 0.50 }
		set_variable = { psLaw_laborSupport = 0.50 }
		set_variable = { psLaw_tariffCode = 0.12 }
		set_variable = { psLaw_tariffSupport = 0.50 }
		set_variable = { psLaw_capitalCode = 0.40 }
		set_variable = { psLaw_capitalSupport = 0.50 }
		set_variable = { psLaw_contractCode = 0.45 }
		set_variable = { psLaw_contractSupport = 0.45 }
		set_variable = { psLaw_soeCode = 0.40 }
		set_variable = { psLaw_soeSupport = 0.45 }
	}
}

# psLaw_inputLaw: 1 tax, 2 labor, 3 tariff, 4 capital, 5 contract, 6 state-firm privilege.
psLaw_shiftSupport = {
	if = {
		limit = { check_variable = { psLaw_inputLaw = 1 } }
		add_to_variable = { psLaw_taxSupport = psLaw_inputStep }
		clamp_variable = { var = psLaw_taxSupport min = 0.05 max = 0.95 }
	}
	else_if = {
		limit = { check_variable = { psLaw_inputLaw = 2 } }
		add_to_variable = { psLaw_laborSupport = psLaw_inputStep }
		clamp_variable = { var = psLaw_laborSupport min = 0.05 max = 0.95 }
	}
	else_if = {
		limit = { check_variable = { psLaw_inputLaw = 3 } }
		add_to_variable = { psLaw_tariffSupport = psLaw_inputStep }
		clamp_variable = { var = psLaw_tariffSupport min = 0.05 max = 0.95 }
	}
	else_if = {
		limit = { check_variable = { psLaw_inputLaw = 4 } }
		add_to_variable = { psLaw_capitalSupport = psLaw_inputStep }
		clamp_variable = { var = psLaw_capitalSupport min = 0.05 max = 0.95 }
	}
	else_if = {
		limit = { check_variable = { psLaw_inputLaw = 5 } }
		add_to_variable = { psLaw_contractSupport = psLaw_inputStep }
		clamp_variable = { var = psLaw_contractSupport min = 0.05 max = 0.95 }
	}
	else_if = {
		limit = { check_variable = { psLaw_inputLaw = 6 } }
		add_to_variable = { psLaw_soeSupport = psLaw_inputStep }
		clamp_variable = { var = psLaw_soeSupport min = 0.05 max = 0.95 }
	}
	force_update_dynamic_modifier = yes
	psEcon_refresh = yes
}
