# Ownership stocks. 1 is state, 0 is private. Laws do not move these.
# Opening shares are history data. psInd_apply writes this month's level effects.

psInd_fallback = {
	if = {
		limit = { NOT = { has_variable = psInd_heavyState } }
		set_variable = { psInd_heavyState = 0.40 }
		set_variable = { psInd_lightState = 0.35 }
		set_variable = { psInd_techState = 0.30 }
		set_variable = { psInd_farmState = 0.35 }
		set_variable = { psInd_energyState = 0.55 }
	}
}

# Private share is the part the state does not hold.
psInd_mirrorPrivate = {
	set_variable = { psInd_heavyPrivate = 1 }
	subtract_from_variable = { psInd_heavyPrivate = psInd_heavyState }
	set_variable = { psInd_lightPrivate = 1 }
	subtract_from_variable = { psInd_lightPrivate = psInd_lightState }
	set_variable = { psInd_techPrivate = 1 }
	subtract_from_variable = { psInd_techPrivate = psInd_techState }
	set_variable = { psInd_farmPrivate = 1 }
	subtract_from_variable = { psInd_farmPrivate = psInd_farmState }
	set_variable = { psInd_energyPrivate = 1 }
	subtract_from_variable = { psInd_energyPrivate = psInd_energyState }
}

# Easy money plus private tech and light industry froths the stock. State heavy and energy lean against it.
psInd_bubblePressure = {
	set_temp_variable = { easy = psEcon_stanceUsed }
	clamp_temp_variable = { var = easy min = 0 max = 2 }
	set_temp_variable = { froth = psInd_techPrivate }
	multiply_temp_variable = { froth = psInd_lightPrivate }
	multiply_temp_variable = { froth = easy }
	multiply_temp_variable = { froth = 0.0035 }
	set_temp_variable = { runup = psEcon_momentum }
	clamp_temp_variable = { var = runup min = 0 max = 0.04 }
	multiply_temp_variable = { runup = psInd_techPrivate }
	multiply_temp_variable = { runup = 0.12 }
	add_to_temp_variable = { froth = runup }
	set_temp_variable = { anchor = psInd_heavyState }
	multiply_temp_variable = { anchor = 0.0012 }
	subtract_from_temp_variable = { froth = anchor }
	set_temp_variable = { anchor = psInd_energyState }
	multiply_temp_variable = { anchor = 0.0008 }
	subtract_from_temp_variable = { froth = anchor }
	set_variable = { psInd_bubbleAdd = froth }
}

# psInd_inputSector: 1 heavy, 2 light, 3 tech, 4 farm, 5 energy. Positive step nationalizes.
psInd_shift = {
	if = {
		limit = { check_variable = { psInd_inputSector = 1 } }
		add_to_variable = { psInd_heavyState = psInd_inputStep }
		clamp_variable = { var = psInd_heavyState min = 0.02 max = 0.98 }
	}
	else_if = {
		limit = { check_variable = { psInd_inputSector = 2 } }
		add_to_variable = { psInd_lightState = psInd_inputStep }
		clamp_variable = { var = psInd_lightState min = 0.02 max = 0.98 }
	}
	else_if = {
		limit = { check_variable = { psInd_inputSector = 3 } }
		add_to_variable = { psInd_techState = psInd_inputStep }
		clamp_variable = { var = psInd_techState min = 0.02 max = 0.98 }
	}
	else_if = {
		limit = { check_variable = { psInd_inputSector = 4 } }
		add_to_variable = { psInd_farmState = psInd_inputStep }
		clamp_variable = { var = psInd_farmState min = 0.02 max = 0.98 }
	}
	else_if = {
		limit = { check_variable = { psInd_inputSector = 5 } }
		add_to_variable = { psInd_energyState = psInd_inputStep }
		clamp_variable = { var = psInd_energyState min = 0.02 max = 0.98 }
	}
	psEcon_refresh = yes
}
