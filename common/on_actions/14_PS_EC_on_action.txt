# Economy clock. Startup fills missing stocks, and the books close monthly.
on_actions = {
	on_startup = {
		# Every state gets a stock, then every tag builds its books from history or the fallback.
		effect = {
			every_state = {
				psEcon_ensureState = yes
			}
			every_country = {
				psEcon_bootstrap = yes
			}
		}
	}
	on_daily = {
		# The player refreshes the old population readout. AI does that on the monthly tick.
		effect = {
			if = {
				limit = { is_ai = no }
				EC_ccp = yes
				refresh_education_piechart = yes
				DATE_COUNTRY_GET_EFFECTS = yes
			}
		}
	}
	on_monthly = {
		# Score, pay, and grow the land held this month. AI refreshes the old readout here instead of daily.
		effect = {
			psEcon_monthly = yes
			if = {
				limit = { is_ai = yes }
				EC_ccp = yes
				refresh_education_piechart = yes
			}
			if = {
				limit = { original_tag = ZZZ }
				date_ECOA = yes
			}
		}
	}
}
