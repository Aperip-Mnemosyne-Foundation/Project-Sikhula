# Standing laws. Support changes the bite. Industry ownership is not in this file.
# A dynamic modifier reads these with mtth:. Nothing here is a drift speed.

psLawMtth_taxEnforce = {
	# High support means the code on the books is actually collected.
	base = 0.70
	modifier = {
		set_temp_variable = { hold = psLaw_taxSupport }
		multiply_temp_variable = { hold = 0.30 }
		add = hold
	}
}
